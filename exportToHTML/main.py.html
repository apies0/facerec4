main.py 
1    import cv2
2    import face_recognition
3    import numpy as np
4    import os
5    
6    IMAGES_FOLDER = "images"
7    THRESHOLD = 0.55  # Lower = Stricter, Higher = Looser
8    
9    print("Loading training data from folder...")
10   
11   known_face_encodings = []
12   known_face_names = []
13   
14   if not os.path.exists(IMAGES_FOLDER):
15       print(f"Error: Folder '{IMAGES_FOLDER}' not found.")
16       exit()
17   
18   for filename in os.listdir(IMAGES_FOLDER):
19       if filename.lower().endswith((".jpg", ".jpeg", ".png")):
20           path = os.path.join(IMAGES_FOLDER, filename)
21   
22           try:
23               cv_img = cv2.imread(path)
24   
25               if cv_img is None:
26                   print(f"Warning: Could not read {filename}. Is it corrupted?")
27                   continue
28   
29               rgb_img = cv2.cvtColor(cv_img, cv2.COLOR_BGR2RGB)
30   
31               rgb_img = rgb_img.astype('uint8')
32   
33               encodings = face_recognition.face_encodings(rgb_img)
34   
35               if len(encodings) > 0:
36                   known_face_encodings.append(encodings[0])
37   
38                   # Extract name (everything before the first underscore or number)
39                   # Example: "hafiz1.jpg" -> "hafiz"
40                   raw_name = os.path.splitext(filename)[0]
41                   # Remove digits if you named them hafiz1, hafiz2...
42                   name = ''.join([i for i in raw_name if not i.isdigit()]).strip("_")
43   
44                   known_face_names.append(name)
45                   print(f"Loaded: {filename} -> Labelled as: {name}")
46               else:
47                   print(f"Warning: No face found in {filename}. Skipping.")
48   
49           except Exception as e:
50               print(f"ERROR processing {filename}: {e}")
51   
52   print(f"\nSystem Ready. {len(known_face_names)} faces learned. Press 'q' to quit.")
53   
54   # --- INITIALIZE WEBCAM ---
55   video_capture = cv2.VideoCapture(1)
56   
57   while True:
58       ret, frame = video_capture.read()
59       if not ret:
60           print("Error: Webcam disconnected.")
61           break
62   
63       # 1. Optimization: Resize to 1/4 size for speed
64       small_frame = cv2.resize(frame, (0, 0), fx=0.25, fy=0.25)
65   
66       # 2. Convert to RGB and FORCE uint8 type (Crucial for stability)
67       rgb_small_frame = cv2.cvtColor(small_frame, cv2.COLOR_BGR2RGB)
68       rgb_small_frame = np.ascontiguousarray(rgb_small_frame, dtype=np.uint8)
69   
70       # 3. Find faces
71       face_locations = face_recognition.face_locations(rgb_small_frame)
72       face_encodings = face_recognition.face_encodings(rgb_small_frame, face_locations)
73   
74       face_names = []
75       for face_encoding in face_encodings:
76           # Distance calculation
77           face_distances = face_recognition.face_distance(known_face_encodings, face_encoding)
78           best_match_index = np.argmin(face_distances)
79   
80           # Threshold check
81           if face_distances[best_match_index] < THRESHOLD:
82               name = known_face_names[best_match_index]
83           else:
84               name = "Unknown"
85   
86           face_names.append(name)
87   
88       # 4. Display Results
89       for (top, right, bottom, left), name in zip(face_locations, face_names):
90           # Scale back up (since we resized by 1/4)
91           top *= 4;
92           right *= 4;
93           bottom *= 4;
94           left *= 4
95   
96           color = (0, 255, 0) if name != "Unknown" else (0, 0, 255)
97   
98           cv2.rectangle(frame, (left, top), (right, bottom), color, 2)
99           cv2.rectangle(frame, (left, bottom - 35), (right, bottom), color, cv2.FILLED)
100          cv2.putText(frame, name, (left + 6, bottom - 6), cv2.FONT_HERSHEY_DUPLEX, 0.8, (255, 255, 255), 1)
101  
102      cv2.imshow('Face Recognition System', frame)
103  
104      if cv2.waitKey(1) & 0xFF == ord('q'):
105          break
106  
107  video_capture.release()
108  cv2.destroyAllWindows()
